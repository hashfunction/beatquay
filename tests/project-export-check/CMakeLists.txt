cmake_minimum_required(VERSION 3.24)
project(BeatQuayPureExportChecks LANGUAGES CXX)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Test)
include(CTest)
include(GenerateExportHeader)
set(CMAKE_AUTOMOC ON)
get_filename_component(BEATQUAY_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(ProjectExportCheckCore STATIC "${BEATQUAY_SOURCE}/src/core/ProjectExportCheck.cpp")
generate_export_header(ProjectExportCheckCore BASE_NAME lmms)
target_include_directories(ProjectExportCheckCore PUBLIC "${BEATQUAY_SOURCE}/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(ProjectExportCheckCore PUBLIC Qt6::Core)
target_compile_features(ProjectExportCheckCore PUBLIC cxx_std_20)
add_executable(ProjectExportCheckTest "${BEATQUAY_SOURCE}/tests/src/core/ProjectExportCheckTest.cpp")
target_link_libraries(ProjectExportCheckTest PRIVATE ProjectExportCheckCore Qt6::Test)
if(MSVC)
    target_compile_options(ProjectExportCheckCore PRIVATE /W4 /WX /utf-8)
    target_compile_options(ProjectExportCheckTest PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(ProjectExportCheckCore PRIVATE -Wall -Wextra -Werror)
    target_compile_options(ProjectExportCheckTest PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME ProjectExportCheckTest COMMAND ProjectExportCheckTest)
