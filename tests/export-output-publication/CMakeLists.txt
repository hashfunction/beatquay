cmake_minimum_required(VERSION 3.24)
project(BeatQuayExportOutputPublication LANGUAGES CXX)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Test)
include(CTest)
include(GenerateExportHeader)
set(CMAKE_AUTOMOC ON)
get_filename_component(BEATQUAY_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(ExportOutputPublicationCore STATIC
    "${BEATQUAY_SOURCE}/src/core/ExportFileIdentity.cpp"
    "${BEATQUAY_SOURCE}/src/core/ExportOutputPublication.cpp")
generate_export_header(ExportOutputPublicationCore BASE_NAME lmms)
target_include_directories(ExportOutputPublicationCore PUBLIC "${BEATQUAY_SOURCE}/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(ExportOutputPublicationCore PUBLIC Qt6::Core)
target_compile_features(ExportOutputPublicationCore PUBLIC cxx_std_20)
add_executable(ExportOutputPublicationTest "${BEATQUAY_SOURCE}/tests/src/core/ExportOutputPublicationTest.cpp")
target_link_libraries(ExportOutputPublicationTest PRIVATE ExportOutputPublicationCore Qt6::Test)
if(MSVC)
    target_compile_options(ExportOutputPublicationCore PRIVATE /W4 /WX /utf-8)
    target_compile_options(ExportOutputPublicationTest PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(ExportOutputPublicationCore PRIVATE -Wall -Wextra -Werror)
    target_compile_options(ExportOutputPublicationTest PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME ExportOutputPublicationTest COMMAND ExportOutputPublicationTest)
