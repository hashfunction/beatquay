cmake_minimum_required(VERSION 3.24)
project(BeatQuayAudioFileOutput LANGUAGES CXX)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Test)
include(CTest)
include(GenerateExportHeader)
set(CMAKE_AUTOMOC ON)
get_filename_component(BEATQUAY_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(AudioFileOutputCore STATIC "${BEATQUAY_SOURCE}/src/core/audio/AudioFileOutput.cpp" "${BEATQUAY_SOURCE}/src/core/ExportFileIdentity.cpp")
generate_export_header(AudioFileOutputCore BASE_NAME lmms)
target_include_directories(AudioFileOutputCore PUBLIC "${BEATQUAY_SOURCE}/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(AudioFileOutputCore PUBLIC Qt6::Core)
target_compile_features(AudioFileOutputCore PUBLIC cxx_std_20)
add_executable(AudioFileOutputTest "${BEATQUAY_SOURCE}/tests/src/core/AudioFileOutputTest.cpp")
target_link_libraries(AudioFileOutputTest PRIVATE AudioFileOutputCore Qt6::Test)
if(MSVC)
    target_compile_options(AudioFileOutputCore PRIVATE /W4 /WX /utf-8)
    target_compile_options(AudioFileOutputTest PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(AudioFileOutputCore PRIVATE -Wall -Wextra -Werror)
    target_compile_options(AudioFileOutputTest PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME AudioFileOutputTest COMMAND AudioFileOutputTest)
