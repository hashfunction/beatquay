# Product packaging identity is independent from the upstream project schema.
SET(CPACK_PACKAGE_NAME                          "${BEATQUAY_NAME}")
SET(CPACK_PACKAGE_DESCRIPTION_SUMMARY           "${BEATQUAY_NAME} - Music creation")
SET(CPACK_PACKAGE_VENDOR                        "${BEATQUAY_PUBLISHER}")
SET(CPACK_PACKAGE_DESCRIPTION_FILE              "${CMAKE_SOURCE_DIR}/README.md")
SET(CPACK_RESOURCE_FILE_LICENSE                 "${CMAKE_SOURCE_DIR}/LICENSE.txt")
SET(CPACK_PACKAGE_VERSION                       "${BEATQUAY_VERSION}")
SET(CPACK_PACKAGE_VERSION_MAJOR                 "${BEATQUAY_VERSION_MAJOR}")
SET(CPACK_PACKAGE_VERSION_MINOR                 "${BEATQUAY_VERSION_MINOR}")
SET(CPACK_PACKAGE_VERSION_PATCH                 "${BEATQUAY_VERSION_PATCH}")
SET(CPACK_PACKAGE_INSTALL_DIRECTORY             "${BEATQUAY_NAME}")
SET(CPACK_SOURCE_GENERATOR                      "TBZ2")
SET(CPACK_SOURCE_PACKAGE_FILE_NAME              "${BEATQUAY_NAME}-${BEATQUAY_VERSION}")
SET(CPACK_PACKAGE_EXECUTABLES                   "${CMAKE_PROJECT_NAME}" "${BEATQUAY_NAME}")

# Disable strip for Debug|RelWithDebInfo
if(CMAKE_BUILD_TYPE MATCHES "Deb")
	unset(CPACK_STRIP_FILES)
else()
	set(CPACK_STRIP_FILES TRUE)
endif()

IF(LMMS_BUILD_WIN32)
	ADD_SUBDIRECTORY(nsis)
ELSEIF(LMMS_BUILD_APPLE)
	ADD_SUBDIRECTORY(apple)
ELSE()
	ADD_SUBDIRECTORY(linux)
ENDIF()

INCLUDE(CPack)
